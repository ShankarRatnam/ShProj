Google Sheets Auto-Sync Setup

Link this app to your Travel Expense Tracker (INR) Google Sheet so every new expense is appended automatically.

One-Time Setup (Takes 60 seconds in your Google Sheet):

  1. Open your Travel Expense Tracker (INR) sheet and click Extensions → Apps Script.
  2. Replace any code in the editor with the script below and click Save (💾):
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Expenses');
  var data = JSON.parse(e.postData.contents);
  var rows = Array.isArray(data.rows) ? data.rows : [data];
  rows.forEach(function(r) {
    sheet.appendRow([r.travelName, r.category, r.detail, r.dateTime, Number(r.amount)]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
